local function save(inst, prop, value)
	local e = store[inst]
	if not e then
		e = { __has = {} }
		store[inst] = e
	end
	if not e.__has[prop] then
		local ok, old = pcall(function() return inst[prop] end)
		if not ok then return end
		e[prop] = old
		e.__has[prop] = true
	end
	pcall(function() inst[prop] = value end)
end

local function restoreAll(done)
	local list = {}
	for inst, props in pairs(store) do
		list[#list + 1] = { inst, props }
	end
	store = {}

	local i = 1
	task.spawn(function()
		while i <= #list do
			local budget = 400
			while i <= #list and budget > 0 do
				local entry = list[i]
				i = i + 1
				budget = budget - 1
				local inst, props = entry[1], entry[2]
				if typeof(inst) == "Instance" then
					for prop, val in pairs(props) do
						if prop ~= "__has" then
							pcall(function() inst[prop] = val end)
						end
					end
				end
			end
			task.wait()
		end
		if done then done() end
	end)
end

local function greyOf(c)
	local lum = 0.2126 * c.R + 0.7152 * c.G + 0.0722 * c.B
	lum = math.clamp(lum, 0, 1)
	local g = 0.26 + lum * 0.40
	return Color3.new(g, g, g)
end

local HIDE_TOKENS = {"grass","foliage","bush","weed","leaf","shrub","plant","fern","flower"}
local KEEP_TOKENS = {"ground","floor","baseplate","wall","road","terrain","slab"}

local function isVegetation(inst, material)
	if material == Enum.Material.Grass or material == Enum.Material.LeafyGrass then
		return true
	end
	local n = string.lower(inst.Name)
	for _, t in ipairs(HIDE_TOKENS) do
		if string.find(n, t, 1, true) then
			for _, k in ipairs(KEEP_TOKENS) do
				if string.find(n, k, 1, true) then return false end
			end
			return true
		end
	end
	return false
end

local function process(inst)
	if not active then return end
	local cls = inst.ClassName

	if inst:IsA("BasePart") then
		local mat = inst.Material
		if mat ~= Enum.Material.Neon and mat ~= Enum.Material.ForceField then
			save(inst, "Material", Enum.Material.SmoothPlastic)
		end
		save(inst, "Color", greyOf(inst.Color))
		save(inst, "Reflectance", 0)
		save(inst, "CastShadow", false)

		if cls == "MeshPart" then
			if inst.TextureID ~= "" then save(inst, "TextureID", "") end
			local sa = inst:FindFirstChildOfClass("SurfaceAppearance")
			if sa then save(sa, "ColorMap", "") end
		end

		if isVegetation(inst, mat) then
			save(inst, "Transparency", 1)
		end
		optimized = optimized + 1
		return
	end

	if inst:IsA("Decal") then save(inst, "Transparency", 1) return end
	if inst:IsA("Light") then save(inst, "Enabled", false) return end

	local ep = EFFECT_PROP[cls]
	if ep then save(inst, ep, false) return end

	if cls == "Sky" or cls == "Atmosphere" or cls == "Clouds" then
		save(inst, "Parent", nil)
	end
end

local function enqueue(inst)
	if queued[inst] then return end
	queued[inst] = true
	queue[#queue + 1] = inst
end

local function startWorker()
	if worker then return end
	worker = task.spawn(function()
		while active do
			local budget = 220
			while active and qHead <= #queue and budget > 0 do
				local inst = queue[qHead]
				queue[qHead] = nil
				qHead = qHead + 1
				budget = budget - 1
				if inst then
					pcall(process, inst)
					queued[inst] = nil
				end
			end
			if qHead > #queue then
				queue = {}
				qHead = 1
			end
			task.wait()
		end
		worker = nil
	end)
end

local function scan()
	local ok, list = pcall(function() return Workspace:GetDescendants() end)
	if ok and list then
		for i = 1, #list do enqueue(list[i]) end
	end
	local ok2, list2 = pcall(function() return Lighting:GetDescendants() end)
	if ok2 and list2 then
		for i = 1, #list2 do enqueue(list2[i]) end
	end
end

local function aliveChar(plr)
	local c = plr.Character
	if not c or not c.Parent then return nil end
	local h = c:FindFirstChildOfClass("Humanoid")
	if not h or h.Health <= 0 then return nil end
	local hrp = c:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil end
	return c, h, hrp
end

local function teamKey(plr)
	if plr.Team then return "T:" .. plr.Team.Name end

	local a = plr:GetAttribute("Team") or plr:GetAttribute("TeamName") or plr:GetAttribute("team")
	if a ~= nil then return "A:" .. tostring(a) end

	local c = plr.Character
	if c then
		local ca = c:GetAttribute("Team") or c:GetAttribute("TeamName")
		if ca ~= nil then return "A:" .. tostring(ca) end
		local tv = c:FindFirstChild("Team", true)
		if tv and (tv:IsA("StringValue") or tv:IsA("IntValue") or tv:IsA("NumberValue")) then
			return "A:" .. tostring(tv.Value)
		end
	end

	local pv = plr:FindFirstChild("Team")
	if pv and (pv:IsA("StringValue") or pv:IsA("IntValue")) then
		return "A:" .. tostring(pv.Value)
	end
	return nil
end

local function isEnemy(plr)
	if plr == LP then return false end
	if not teamFilter then return true end
	local mine = teamKey(LP)
	local theirs = teamKey(plr)
	if mine == nil or theirs == nil then return true end
	return mine ~= theirs
end

local function inMatch()
	local c = aliveChar(LP)
	if not c then return false end
	local node = c.Parent
	local depth = 0
	while node and node ~= Workspace and depth < 5 do
		local n = string.lower(node.Name)
		if string.find(n, "lobby", 1, true)
			or string.find(n, "intermission", 1, true)
			or string.find(n, "spectat", 1, true) then
			return false
		end
		node = node.Parent
		depth = depth + 1
	end
	return true
end

local function paintHighlight(plr)
	local h = highlights[plr]
	if not h then return end
	local isTarget = (plr == currentTarget) or (autoAim and plr == autoTarget)
	if isTarget then
		h.OutlineColor = TARGET_COL
		h.FillColor = TARGET_COL
		h.FillTransparency = 0.72
	elseif not isEnemy(plr) then
		h.OutlineColor = TEAM_COL
		h.FillColor = TEAM_COL
		h.FillTransparency = 1
	else
		h.OutlineColor = RED
		h.FillColor = RED
		h.FillTransparency = 1
	end
	h.OutlineTransparency = 0
end

local function repaintAll()
	for plr in pairs(highlights) do
		pcall(paintHighlight, plr)
	end
end

local function ensureFolder()
	if espFolder and espFolder.Parent then return espFolder end
	local f = Instance.new("Folder")
	f.Name = "PME_ESP"
	f.Parent = Workspace
	espFolder = f
	return f
end

local function removeHighlight(plr)
	local h = highlights[plr]
	highlights[plr] = nil
	if h then pcall(function() h:Destroy() end) end
end

local function removeAllHighlights()
	for plr in pairs(highlights) do removeHighlight(plr) end
	highlights = {}
	if espFolder then
		pcall(function() espFolder:Destroy() end)
		espFolder = nil
	end
end

local function reconcile()
	if not active then return end
	local folder = ensureFolder()

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then
			local char = aliveChar(plr)
			if char then
				local h = highlights[plr]
				if not h or not h.Parent then
					if h then pcall(function() h:Destroy() end) end
					h = Instance.new("Highlight")
					h.Name = plr.Name
					h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
					h.FillTransparency = 1
					h.OutlineTransparency = 0
					h.OutlineColor = RED
					h.FillColor = RED
					h.Parent = folder
					highlights[plr] = h
					paintHighlight(plr)
				end
				if h.Adornee ~= char then h.Adornee = char end
			else
				if highlights[plr] then removeHighlight(plr) end
			end
		end
	end

	local dead = {}
	for plr in pairs(highlights) do
		if not plr.Parent or plr == LP then dead[#dead + 1] = plr end
	end
	for _, plr in ipairs(dead) do removeHighlight(plr) end
end

local function aimPoint(char)
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if hrp then return hrp.Position + Vector3.new(0, 0.45, 0) end
	local head = char:FindFirstChild("Head")
	if head then return head.Position end
	return nil
end

local function pickEnemy(fromPos, exclude)
	local best, bestDist = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= exclude and isEnemy(plr) then
			local _, _, hrp = aliveChar(plr)
			if hrp then
				local d = (hrp.Position - fromPos).Magnitude
				if d < bestDist then
					best, bestDist = plr, d
				end
			end
		end
	end
	return best
end

local function onRender(dt)
	if not active then return end
	local cam = Workspace.CurrentCamera
	if not cam then return end

	if not matchActive then
		autoTarget = nil
		return
	end

	if currentTarget then
		local ok = aliveChar(currentTarget)
		if not ok or not isEnemy(currentTarget) then
			if autoSwitch then
				currentTarget = pickEnemy(cam.CFrame.Position, currentTarget)
			else
				currentTarget = nil
			end
			rowDirty = true
			task.spawn(repaintAll)
		end
	end

	if autoAim then
		autoAccum = autoAccum + dt
		if autoAccum >= 0.15 then
			autoAccum = 0
			local n = pickEnemy(cam.CFrame.Position, nil)
			if n ~= autoTarget then
				autoTarget = n
				task.spawn(repaintAll)
			end
		end
	end

	local target = currentTarget or autoTarget
	if not target then return end

	local char = aliveChar(target)
	if not char then return end

	local point = aimPoint(char)
	if not point then return end

	local alpha = math.clamp(aimSpeed * dt * 60, 0, 1)

	if cam.CameraType == Enum.CameraType.Scriptable then
		if typeof(mousemoverel) == "function" then
			local pos, onScreen = cam:WorldToViewportPoint(point)
			if onScreen then
				local vp = cam.ViewportSize
				local dx = (pos.X - vp.X * 0.5) * alpha
				local dy = (pos.Y - vp.Y * 0.5) * alpha
				if math.abs(dx) > 0.4 or math.abs(dy) > 0.4 then
					pcall(mousemoverel, dx, dy)
				end
			end
		end
	end

	local desired = CFrame.lookAt(cam.CFrame.Position, point)
	cam.CFrame = cam.CFrame:Lerp(desired, alpha)
end

local function viewport()
	local v = gui.AbsoluteSize
	if v.X < 100 or v.Y < 100 then
		local cam = Workspace.CurrentCamera
		v = cam and cam.ViewportSize or Vector2.new(800, 450)
	end
	return v
end

local function clampTo(x, y, w, h)
	local v = viewport()
	return UDim2.fromOffset(
		math.clamp(x, 0, math.max(0, v.X - w)),
		math.clamp(y, 0, math.max(0, v.Y - h))
	)
end

local dragging, dragStart, startPos = false, nil, nil

local function inIconRect(pos)
	local ap, as = icon.AbsolutePosition, icon.AbsoluteSize
	return pos.X >= ap.X and pos.X <= ap.X + as.X
		and pos.Y >= ap.Y and pos.Y <= ap.Y + as.Y
end

local function bindDrag(target, tapMode)
	target.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if tapMode == "open" and opened then return end
		if tapMode == "close" and not opened then return end
		if tapMode == "drag" and not opened then return end

		dragging = true
		dragStart = input.Position
		startPos = root.Position
		local moved = 0

		local changed
		changed = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
				if changed then changed:Disconnect() end
				if moved < 8 then
					if tapMode == "open" then setOpen(true)
					elseif tapMode == "close" then setOpen(false) end
				end
				return
			end

			if input.UserInputType == Enum.UserInputType.MouseMovement
				or input.UserInputType == Enum.UserInputType.Touch then
				local d = input.Position - dragStart
				moved = math.max(moved, d.Magnitude)
				local w = root.Size.X.Offset
				local h = root.Size.Y.Offset
				root.Position = clampTo(startPos.X.Offset + d.X, startPos.Y.Offset + d.Y, w, h)
			end
		end)
	end)
end

bindDrag(root, "open")
bindDrag(icon, "close")
bindDrag(header, "drag")

-- ENABLE / DISABLE

local function enable()
	if active or restoring then return end
	active = true
	optimized = 0

	pcall(applyWorld)

	table.insert(conns, Workspace.DescendantAdded:Connect(function(d)
		if active then enqueue(d) end
	end))
	table.insert(conns, Lighting.DescendantAdded:Connect(function(d)
		if active then enqueue(d) end
	end))

	scan()
	startWorker()
	reconcile()

	pcall(function()
		RunService:UnbindFromRenderStep("PME_AIM")
		RunService:BindToRenderStep("PME_AIM", Enum.RenderPriority.Camera.Value + 50, onRender)
	end)
end

local function disable(finished)
	if not active then
		if finished then finished() end
		return
	end
	active = false
	autoTarget = nil
	matchActive = false

	pcall(function() RunService:UnbindFromRenderStep("PME_AIM") end)
	removeAllHighlights()

	for i = 1, #conns do pcall(function() conns[i]:Disconnect() end) end
	conns = {}
	queue = {}
	qHead = 1
	queued = {}

	optimized = 0
	restoring = true
	restoreAll(function()
		restoring = false
		if finished then finished() end
	end)
end

local function setSpeed(v)
	aimSpeed = math.clamp(v, 0.05, 1)
	speedKnob.Position = UDim2.new(aimSpeed, 0, 0.5, 0)
	speedFill.Size = UDim2.new(aimSpeed, 0, 1, 0)
	speedVal.Text = ("%d%%"):format(math.floor(aimSpeed * 100))
end

local function speedFromX(x)
	local w = math.max(speedTrack.AbsoluteSize.X, 1)
	setSpeed((x - speedTrack.AbsolutePosition.X) / w)
end

speedTrack.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
		speedFromX(input.Position.X)

		local changed
		changed = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				if changed then changed:Disconnect() end
				return
			end
			if input.UserInputType == Enum.UserInputType.MouseMovement
				or input.UserInputType == Enum.UserInputType.Touch then
				speedFromX(input.Position.X)
			end
		end)
	end
end)

local function watchPlayer(plr)
	if plr == LP or playerWatches[plr] then return end

	local c1 = plr.CharacterAdded:Connect(function()
		if active then task.defer(reconcile) end
		rowDirty = true
		if plr == currentTarget then repaintAll() end
	end)
	local c2 = plr.CharacterRemoving:Connect(function()
		removeHighlight(plr)
	end)
	playerWatches[plr] = { c1, c2 }

	if active then pcall(reconcile) end
	rowDirty = true
end

local function unwatchPlayer(plr)
	local w = playerWatches[plr]
	if w then
		for i = 1, #w do pcall(function() w[i]:Disconnect() end) end
		playerWatches[plr] = nil
	end
	removeHighlight(plr)
end

for _, plr in ipairs(Players:GetPlayers()) do watchPlayer(plr) end

Players.PlayerAdded:Connect(watchPlayer)
Players.PlayerRemoving:Connect(function(plr)
	unwatchPlayer(plr)
	if currentTarget == plr then currentTarget = nil end
	if autoTarget == plr then autoTarget = nil end
	rowDirty = true
end)

LP.CharacterAdded:Connect(function()
	matchActive = inMatch()
	rowDirty = true
end)
LP.CharacterRemoving:Connect(function()
	autoTarget = nil
end)
